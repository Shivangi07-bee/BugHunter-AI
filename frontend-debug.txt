import { useState } from "react";

import Navbar from "../components/Navbar";
import RepositoryUploader from "../components/RepositoryUploader";


function Dashboard() {

    const [targetName, setTargetName] = useState("");
    const [targetUrl, setTargetUrl] = useState("");
    const [targetScope, setTargetScope] = useState("");

    const [loading, setLoading] = useState(false);
    const [error, setError] = useState("");


    const handleTargetAnalyze = async () => {

        setError("");

        if (!targetUrl.trim()) {
            setError("Target URL is required.");
            return;
        }

        setLoading(true);

        try {

            const response = await fetch(
                "http://localhost:8000/target/context",
                {
                    method: "POST",

                    headers: {
                        "Content-Type": "application/json",
                    },

                    body: JSON.stringify({
                        name:
                            targetName.trim() ||
                            targetUrl.trim(),

                        base_url:
                            targetUrl.trim(),

                        scope:
                            targetScope.trim()
                                ? [targetScope.trim()]
                                : [],

                        source: "authorized_target",
                    }),
                }
            );


            if (!response.ok) {

                let message =
                    "Unable to validate target.";

                try {
                    const data = await response.json();

                    if (data?.detail) {
                        message = data.detail;
                    }
                } catch {
                    // Keep default error message
                }

                throw new Error(message);
            }


            const target = await response.json();


            /*
             * Persist the validated target.
             * Investigation page will consume this context.
             */

            localStorage.setItem(
                "bughunter_target",
                JSON.stringify(target)
            );


            /*
             * Move into the investigation workspace.
             */

            window.location.href =
                "/investigation";


        } catch (err) {

            console.error(
                "Target initialization failed:",
                err
            );

            setError(
                err?.message ||
                "Unable to initialize investigation."
            );

        } finally {

            setLoading(false);

        }
    };

        setTargetName("");
        setTargetUrl("");
        setTargetScope("");
        setError("");

    };

                                "0 12px 40px rgba(0, 0, 0, 0.25)",
                        }}
                    >

                                gap: "20px",
                                marginBottom: "26px",
                            }}
                        >

                            <div>

                                <div
                                    style={{
                                        fontSize: "11px",
                                        letterSpacing: "2px",
                                        color: "#20cfff",
                                        fontWeight: 700,
                                        marginBottom: "8px",
                                    }}
                                >
                                    SECURITY TARGET
                                </div>


                                <h1
                                    style={{
                                        margin: 0,
